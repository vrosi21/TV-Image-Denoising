cmake_minimum_required(VERSION 3.17)

set(SOLUTION_NAME TVDenoising)

project(${SOLUTION_NAME})
set(SOURCE_ROOT ${CMAKE_CURRENT_LIST_DIR})

if (WIN32)
    set(HOME_ROOT $ENV{USERPROFILE})
    string(REPLACE "\\" "/" HOME_ROOT "${HOME_ROOT}")
else()
    set(HOME_ROOT $ENV{HOME})
endif()

set(NATID_SDK_ROOT ${HOME_ROOT}/natID.SDK)
include(${NATID_SDK_ROOT}/DevEnv/Common.cmake)
include(${NATID_SDK_ROOT}/DevEnv/natGUI.cmake)
include(${NATID_SDK_ROOT}/DevEnv/MatrixLib.cmake)
include(TVDenoising.cmake)

# ── CPack installer configuration ─────────────────────────────────
set(CPACK_PACKAGE_NAME              "TVDenoising")
set(CPACK_PACKAGE_VENDOR            "Kemal Sivro - ETF Sarajevo")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "TV Image Denoising — Gradient Descent & Newton's Method GUI")
set(CPACK_PACKAGE_VERSION           "1.0.0")
set(CPACK_PACKAGE_VERSION_MAJOR     "1")
set(CPACK_PACKAGE_VERSION_MINOR     "0")
set(CPACK_PACKAGE_VERSION_PATCH     "0")
set(CPACK_PACKAGE_INSTALL_DIRECTORY "TVDenoising")
set(CPACK_RESOURCE_FILE_LICENSE     "${CMAKE_CURRENT_LIST_DIR}/LICENSE" )

if (WIN32)
    set(CPACK_GENERATOR "NSIS")
    set(CPACK_NSIS_PACKAGE_NAME     "TV Image Denoising")
    set(CPACK_NSIS_DISPLAY_NAME     "TV Image Denoising")
    set(CPACK_NSIS_HELP_LINK        "https://github.com/vrosi21/TV-Image-Denoising")
    set(CPACK_NSIS_URL_INFO_ABOUT   "https://github.com/vrosi21/TV-Image-Denoising")
    set(CPACK_NSIS_MUI_ICON         "${CMAKE_CURRENT_LIST_DIR}/res/appIcon/winApp.ico")
    set(CPACK_WIX_UPGRADE_GUID      "B3A7C2D1-4E5F-6A8B-9C0D-1E2F3A4B5C6D")
    set(CPACK_WIX_PRODUCT_ICON      "${CMAKE_CURRENT_LIST_DIR}/res/appIcon/winApp.ico")
elseif (APPLE)
    set(CPACK_GENERATOR "ZIP")
    set(CPACK_PACKAGE_FILE_NAME     "${CPACK_PACKAGE_NAME}-macOS")
elseif (UNIX)
    set(CPACK_GENERATOR "DEB")
    set(CPACK_DEBIAN_PACKAGE_NAME   "tvdenoising")
    set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Kemal Sivro <kemalsivro@gmail.com>")
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "libgtk-3-0, libglib2.0-0, libcairo2")
    set(CPACK_DEBIAN_PACKAGE_SECTION "science")
    set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/vrosi21/TV-Image-Denoising")
endif()

include(CPack)
